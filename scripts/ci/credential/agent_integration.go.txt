type credentialDropResponse struct {
	next    http.RoundTripper
	dropped bool
}

func (transport *credentialDropResponse) RoundTrip(request *http.Request) (*http.Response, error) {
	response, err := transport.next.RoundTrip(request)
	if err != nil {
		return nil, err
	}
	if !transport.dropped {
		transport.dropped = true
		_, _ = io.Copy(io.Discard, response.Body)
		response.Body.Close()
		return nil, errors.New("synthetic response loss after committed backend rotation")
	}
	return response, nil
}

func TestCredentialLifecycleActualAgentLostResponseAndRestart(t *testing.T) {
	record, token, key, _ := refreshFixture(t)
	_, handler := enrollmentconnect.NewEnrollmentServiceHandler(&enrollmentService{})
	server := httptest.NewServer(handler)
	defer server.Close()
	signing, err := enrollment.ActiveSigningKey()
	require.NoError(t, err)
	store, err := agentcredentials.Open(filepath.Join(t.TempDir(), "identity.json"))
	require.NoError(t, err)
	identity := agentcredentials.Identity{Endpoint: server.URL, AgentID: record.Client, PrivateKey: base64.StdEncoding.EncodeToString(key), PublicKey: record.AgentPublicKey, AccessToken: "synthetic-access" + token, RefreshToken: token, AccessTokenExpiresAt: record.AccessExpiresAt, RefreshTokenExpiresAt: record.RefreshExpiresAt, ControlPlaneKeys: []agentcredentials.PinnedKey{{Algorithm: 1, KeyID: signing.KeyID, Value: base64.StdEncoding.EncodeToString(signing.PublicKey)}}}
	require.NoError(t, store.Save(identity))
	pending, err := store.BeginRefresh()
	require.NoError(t, err)
	ctx := context.Background()
	client, err := agentenrollment.New(server.URL, &http.Client{Transport: &credentialDropResponse{next: server.Client().Transport}})
	require.NoError(t, err)
	_, err = client.Refresh(ctx, pending)
	require.Error(t, err)
	var current models.AgentCredential
	require.NoError(t, dbcore.GetDBInstance().First(&current, "client = ?", record.Client).Error)
	require.Equal(t, uint64(1), current.Generation, "server must have committed before network loss")
	reopened, err := agentcredentials.Open(store.Path())
	require.NoError(t, err)
	retry, err := reopened.BeginRefresh()
	require.NoError(t, err)
	require.Equal(t, pending.PendingRefresh.RequestID, retry.PendingRefresh.RequestID)
	recovered, err := client.Refresh(ctx, retry)
	require.NoError(t, err)
	require.NoError(t, reopened.CommitRefresh(retry, recovered))
	require.Equal(t, uint64(1), *recovered.Generation)
	require.NoError(t, dbcore.GetDBInstance().First(&current, "client = ?", record.Client).Error)
	require.Equal(t, uint64(1), current.Generation, "recovery must not rotate again")
	resolved, err := enrollment.ResolveAccessToken(recovered.AccessToken)
	require.NoError(t, err)
	require.Equal(t, record.Client, resolved)
	second, err := reopened.BeginRefresh()
	require.NoError(t, err)
	advanced, err := client.Refresh(ctx, second)
	require.NoError(t, err)
	require.NoError(t, reopened.CommitRefresh(second, advanced))
	require.Equal(t, uint64(2), *advanced.Generation)
	require.NoError(t, enrollment.Revoke(record.Client, "synthetic integration revocation"))
	third, err := reopened.BeginRefresh()
	require.NoError(t, err)
	_, err = client.Refresh(ctx, third)
	require.Equal(t, connect.CodeUnauthenticated, connect.CodeOf(err))
	unchanged, _ := reopened.Identity()
	require.Equal(t, advanced.RefreshToken, unchanged.RefreshToken)
	require.Equal(t, uint64(2), *unchanged.Generation)
}
